RESULTADO: Strato só trabalha durante o encode final (20-40s)
Durante os minutos de edição criativa do utilizador — timeline, filtros, trim — o CPU do Strato está a zero para este job. O pico só ocorre quando o utilizador clica "Render Final".
O EDIT SCRIPT é o protocolo central desta arquitectura
O client envia ~2KB de JSON com todas as decisões criativas. O servidor nunca vê o utilizador editar — só recebe o resultado. É o "script de realização" que o artesão executa.

SHA-256 local Web Crypto API · zero servidor
// Hash do raw file antes de qualquer upload
async function hashFile(file) {
const buf = await file.arrayBuffer();
const hash = await crypto.subtle.digest('SHA-256', buf);
return [...new Uint8Array(hash)]
.map(b => b.toString(16).padStart(2,'0')).join('');
}

build_ffmpeg_command()
def build_ffmpeg_command(s, input_path, output_path):
CROPS = {
"9:16": "crop=ih*9/16:ih:(iw-ih*9/16)/2:0,scale=720:1280",
"1:1": "crop=min(iw\\,ih):min(iw\\,ih),scale=720:720",
"16:9": "scale=1280:720"
}
vf_chain = [CROPS[s["format"]]]
if s.get("lut"):
vf_chain.append(f"lut3d=luts/{s['lut']}.cube")
vf_chain.append("drawtext=text='WINDI·{hash8}':...")
return ["ffmpeg","-i",input_path,
"-ss",str(s["trim"]["start"]),"-to",str(s["trim"]["end"]),
"-vf", ",".join(vf_chain),
"-c:v","libx264","-preset","fast","-crf","28",
"-movflags","+faststart", output_path]

# Máx 2 jobs FFmpeg em paralelo
ffmpeg_semaphore = threading.Semaphore(2)
 
def run_ffmpeg_job(script):
with ffmpeg_semaphore:
db_update_status(script["job_id"], "processing")
cmd = build_ffmpeg_command(script, ...)
result = subprocess.run(cmd, timeout=180)
if result.returncode == 0:
output_hash = sha256_file(output_path)
seal_to_ledger(script, output_hash) # → :8101
db_update_status(script["job_id"], "done")

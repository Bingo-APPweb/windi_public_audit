const loadDocs = async () => {
  try {
    const res = await fetch('/api/communique/list');
    const data = await res.json();
    setDocuments(data.communiques || []);
  } catch (e) {
    console.warn('Communiqué API offline, using cache');
  }
};

useEffect(() => {
  loadDocs();
  const interval = setInterval(loadDocs, 60000);
  return () => clearInterval(interval);
}, []);

📋 CCode Prompt — Task C1 (Pott Engine) 
Cria /opt/windi/pott-engine/ com estrutura:
  pott_engine.py (FastAPI main)
  models.py (Pott, PottMembership, PottDiscoveryIndex)
  database.py (SQLite /opt/windi/data/pott.db)

Porta: 8110

Modelo Pott:
  id: str               # "POTT-{vertical}-{hash8}"
  name: str             # "Bavarian Chefs"
  vertical: str         # cooking | education | music | local | tech
  members: list[str]    # lista de creator DIDs
  curator_did: str      # DID do curador principal
  split_creator: float  # 0.85 (85%)
  split_pott: float     # 0.15 (15%)
  rules: dict           # data boundaries, content policies
  created_at: str
  sealed_hash: str
  ledger_receipt: str

Princípios:
1. Data Boundaries: dados de audiência nunca cruzam entre criadores
2. Attribution 85/15: toda receita via Pott: 85% criador, 15% treasury
3. Exit Freedom: criador sai quando quiser, leva todo o conteúdo
4. Receipts: toda acção gera Virtue Receipt no Ledger (:8101)

Cria systemd unit: windi-pott.service
Configura nginx: /api/pott/ → proxy_pass http://127.0.0.1:8110

import sqlite3
from pathlib import Path

METRICS_DB = Path("/opt/windi/windi-gateway/gateway_metrics.db")

def init_metrics_db():
    """Cria tabela de métricas se não existir."""
    conn = sqlite3.connect(METRICS_DB)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS calls (
            id          TEXT PRIMARY KEY,
            ts          TEXT NOT NULL,
            actor       TEXT,
            agent       TEXT,
            task        TEXT,
            provider    TEXT,
            tier        TEXT,
            tokens      INTEGER DEFAULT 0,
            cost_eur    REAL    DEFAULT 0.0,
            pii_blocked INTEGER DEFAULT 0,
            bunker_was_active INTEGER DEFAULT 0,
            duration_ms INTEGER DEFAULT 0,
            week        TEXT    -- YYYY-WNN para groupBy semanal
        )
    """)
    conn.commit()
    conn.close()

def record_metric(rid, actor, agent, task, provider,
                  tier, tokens, cost_eur, pii_blocked,
                  bunker_active, duration_ms):
    """Regista uma chamada na DB de métricas."""
    try:
        week = datetime.utcnow().strftime("%Y-W%V")
        conn = sqlite3.connect(METRICS_DB)
        conn.execute("""
            INSERT OR IGNORE INTO calls
            (id,ts,actor,agent,task,provider,tier,tokens,
             cost_eur,pii_blocked,bunker_was_active,duration_ms,week)
            VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?)
        """, (rid, datetime.utcnow().isoformat(), actor,
              agent or "direct", task, provider, tier,
              tokens, cost_eur, pii_blocked,
              1 if bunker_active else 0, duration_ms, week))
        conn.commit()
        conn.close()
    except Exception as e:
        log.warning(f"[METRICS] record failed (non-blocking): {e}")

# ── GET /gateway/report/week ──────────────────────────────
@app.get("/gateway/report/week")
def weekly_report(week: str = None,
                  _auth: str = Depends(verify_secret)):
    """
    Relatório semanal para aprovação do Human Dragon.
    week: formato YYYY-WNN (ex: 2026-W14)
          omitir = semana actual
    I9: relatório gerado pela IA, aprovado pelo Human Dragon.
    """
    target_week = week or datetime.utcnow().strftime("%Y-W%V")
    conn = sqlite3.connect(METRICS_DB)

    # Totais gerais
    totals = conn.execute("""
        SELECT
            COUNT(*)            as total_calls,
            SUM(tokens)         as total_tokens,
            ROUND(SUM(cost_eur),4) as total_cost_eur,
            SUM(pii_blocked)    as total_pii_blocked,
            SUM(bunker_was_active) as bunker_activations
        FROM calls WHERE week = ?
    """, (target_week,)).fetchone()

    # Por provider
    by_provider = conn.execute("""
        SELECT provider,
               COUNT(*) as calls,
               SUM(tokens) as tokens,
               ROUND(SUM(cost_eur),4) as cost_eur
        FROM calls WHERE week = ?
        GROUP BY provider ORDER BY cost_eur DESC
    """, (target_week,)).fetchall()

    # Por agente
    by_agent = conn.execute("""
        SELECT agent,
               COUNT(*) as calls,
               SUM(tokens) as tokens,
               ROUND(SUM(cost_eur),4) as cost_eur,
               SUM(pii_blocked) as pii_blocked
        FROM calls WHERE week = ?
        GROUP BY agent ORDER BY tokens DESC
    """, (target_week,)).fetchall()

    # Por tier
    by_tier = conn.execute("""
        SELECT tier, COUNT(*) as calls,
               SUM(tokens) as tokens,
               ROUND(SUM(cost_eur),4) as cost_eur
        FROM calls WHERE week = ?
        GROUP BY tier
    """, (target_week,)).fetchall()

    # Top tarefas
    top_tasks = conn.execute("""
        SELECT task, COUNT(*) as calls, SUM(tokens) as tokens
        FROM calls WHERE week = ?
        GROUP BY task ORDER BY calls DESC LIMIT 5
    """, (target_week,)).fetchall()

    conn.close()

    # Sovereignty score
    total_calls  = totals[0] or 0
    total_tokens = totals[1] or 0
    local_calls  = next((r[1] for r in by_provider
                         if r[0]=="local"), 0)
    sov_score = round((local_calls/total_calls*100),1) if total_calls else 0

    return {
        "week":     target_week,
        "approval": "PENDING_HUMAN_DRAGON",   # I9
        "summary": {
            "total_calls":      total_calls,
            "total_tokens":     total_tokens,
            "total_cost_eur":   totals[2] or 0,
            "pii_blocked":      totals[3] or 0,
            "bunker_activations": totals[4] or 0,
            "sovereignty_score": f"{sov_score}% local",
        },
        "by_provider": [
            {"provider":r[0],"calls":r[1],
             "tokens":r[2],"cost_eur":r[3]}
            for r in by_provider
        ],
        "by_agent": [
            {"agent":r[0],"calls":r[1],"tokens":r[2],
             "cost_eur":r[3],"pii_blocked":r[4]}
            for r in by_agent
        ],
        "by_tier": [
            {"tier":r[0],"calls":r[1],
             "tokens":r[2],"cost_eur":r[3]}
            for r in by_tier
        ],
        "top_tasks": [
            {"task":r[0],"calls":r[1],"tokens":r[2]}
            for r in top_tasks
        ],
        "principle": "A IA reporta. O Human Dragon aprova. O WINDI regista.",
        "invariants": ["I9","I11","I12"],
    }

               ROUND(SUM(cost_eur),6) as cost,

        FROM calls WHERE ts >= ?
        GROUP BY provider, tier ORDER BY cost DESC
    """, (since,)).fetchall()

        SELECT id, ts, agent, provider, tokens, cost_eur
        FROM calls ORDER BY ts DESC LIMIT 10
    """).fetchall()

    conn.close()

    return {
        "period":   "últimas 24h",
        "since":    since,
        "breakdown": [
            {"provider":r[0],"tier":r[1],"calls":r[2],
             "tokens":r[3],"cost_eur":r[4],"avg_ms":round(r[5],0)}
            for r in rows
        ],
        "recent_calls": [
            {"id":r[0],"ts":r[1],"agent":r[2],
             "provider":r[3],"tokens":r[4],"cost_eur":r[5]}
            for r in recent
        ],
        "bunker": bunker.status(),
    }

"""
Script de relatório semanal — W-GATEWAY-001
Executa via cron toda segunda-feira às 08:00
Envia sumário ao Human Dragon para aprovação (I9)
"""
import sqlite3, smtplib, os
from datetime import datetime
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from dotenv import load_dotenv
from pathlib import Path

load_dotenv()

METRICS_DB   = Path("/opt/windi/windi-gateway/gateway_metrics.db")
SMTP_HOST    = os.getenv("SMTP_HOST","smtp.strato.de")
SMTP_PORT    = int(os.getenv("SMTP_PORT","587"))
SMTP_USER    = os.getenv("SMTP_USER","")
SMTP_PASS    = os.getenv("SMTP_PASS","")
DRAGON_EMAIL = os.getenv("DRAGON_EMAIL","info@a4desk.de")
BASE_URL     = "https://windi-domain.com"
SECRET       = os.getenv("GATEWAY_SECRET","")

def generate_report():
    week = datetime.utcnow().strftime("%Y-W%V")
    conn = sqlite3.connect(METRICS_DB)

    totals = conn.execute("""
        SELECT COUNT(*),SUM(tokens),ROUND(SUM(cost_eur),4),
               SUM(pii_blocked),SUM(bunker_was_active)
        FROM calls WHERE week=?
    """, (week,)).fetchone()

    by_provider = conn.execute("""
        SELECT provider,COUNT(*),SUM(tokens),ROUND(SUM(cost_eur),4)
        FROM calls WHERE week=? GROUP BY provider ORDER BY cost_eur DESC
    """, (week,)).fetchall()

    by_agent = conn.execute("""
        SELECT agent,COUNT(*),SUM(tokens),ROUND(SUM(cost_eur),4)
        FROM calls WHERE week=? GROUP BY agent ORDER BY tokens DESC LIMIT 5
    """, (week,)).fetchall()

    conn.close()

    total_calls = totals[0] or 0
    local_calls = next((r[1] for r in by_provider if r[0]=="local"),0)
    sov = round((local_calls/total_calls*100),1) if total_calls else 0

    return {
        "week":week,"total_calls":total_calls,
        "total_tokens":totals[1] or 0,
        "total_cost":totals[2] or 0,
        "pii_blocked":totals[3] or 0,
        "bunker_events":totals[4] or 0,
        "sovereignty":sov,
        "by_provider":by_provider,
        "by_agent":by_agent,
    }

    )

    approve_url = (f"{BASE_URL}/gateway/report/approve"
                   f"?week={r['week']}&approved=true"
                   f"&secret={SECRET}")
    reject_url  = (f"{BASE_URL}/gateway/report/approve"
                   f"?week={r['week']}&approved=false"
                   f"&secret={SECRET}")

╔══════════════════════════════════════════════════╗
║  W-GATEWAY-001 · Relatório Semanal              ║
║  Liga IA+H · Kempten {datetime.utcnow().year}                    ║
╚══════════════════════════════════════════════════╝

Semana:         {r['week']}
Soberania:      {r['sovereignty']}% local ({'✅' if r['sovereignty'] >= 90 else '⚠️'})
Total chamadas: {r['total_calls']}

{provider_rows}

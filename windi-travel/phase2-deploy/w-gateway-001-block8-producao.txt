/**
 * WINDI Travel · Maria Voice Engine
 * Web Speech API — transcrição LOCAL no device
 * I1: áudio NUNCA sai do device · zero custo · zero dependência
 *
 * Uso:
 *   const maria = new MariaVoice('de-DE');
 *   maria.onResult = (text) => sendToGateway(text);
 *   maria.start();
 */
class MariaVoice {

  constructor(lang = 'de-DE') {
    this.lang        = lang;
    this.recognition = null;
    this.isListening = false;
    this.onResult    = null;   // callback com texto transcrito
    this.onStart     = null;   // callback quando começa a ouvir
    this.onEnd       = null;   // callback quando para
    this.onError     = null;   // callback de erro

    // Detectar suporte
    const SpeechRecognition =
      window.SpeechRecognition || window.webkitSpeechRecognition;

    if (!SpeechRecognition) {
      console.warn('[Maria] Web Speech API não suportada · fallback texto');
      this.supported = false;
      return;
    }

    this.supported     = true;
    this.recognition   = new SpeechRecognition();

    // Configuração
    this.recognition.lang          = lang;
    this.recognition.continuous    = false;  // 1 comando de cada vez
    this.recognition.interimResults = true;  // mostrar texto em tempo real
    this.recognition.maxAlternatives = 1;

    // Eventos
    this.recognition.onstart = () => {
      this.isListening = true;
      console.log('[Maria] A ouvir…');
      if (this.onStart) this.onStart();
    };

    this.recognition.onresult = (e) => {
      const transcript = Array.from(e.results)
        .map(r => r[0].transcript)
        .join('');
      const isFinal = e.results[e.results.length - 1].isFinal;

      // Mostrar interim results em tempo real
      if (!isFinal && this.onInterim) this.onInterim(transcript);

      // Resultado final → pipeline
      if (isFinal) {
        console.log('[Maria] Transcrito:', transcript);
        if (this.onResult) this.onResult(transcript.trim());
      }
    };

    this.recognition.onend = () => {
      this.isListening = false;
      if (this.onEnd) this.onEnd();
    };

    this.recognition.onerror = (e) => {
      this.isListening = false;
      console.warn('[Maria] Erro:', e.error);
      // I10 — fallback gracioso
      const msgs = {
        'not-allowed': 'Microfone não autorizado · usa o campo de texto',
        'no-speech':   'Nenhuma voz detectada · tenta novamente',
        'network':     'Sem rede · funciona offline com texto',
        'aborted':     'Cancelado',
      };
      if (this.onError) this.onError(msgs[e.error] || e.error);
    };
  }

  /* Muda língua do reconhecimento */
  setLang(lang) {
    this.lang = lang;
    const map = { pt:'pt-PT', de:'de-DE', en:'en-GB' };
    if (this.recognition) this.recognition.lang = map[lang] || lang;
  }

  /* Inicia captura */
  start() {
    if (!this.supported) {
      if (this.onError) this.onError('fallback-text');
      return;
    }
    if (this.isListening) this.stop();
    try {
      this.recognition.start();
    } catch(e) {
      console.warn('[Maria] start error:', e);
    }
  }

  /* Para captura */
  stop() {
    if (this.recognition && this.isListening) {
      this.recognition.stop();
    }
  }

  /* Toggle para botão único */
  toggle() {
    if (this.isListening) this.stop();
    else this.start();
  }
}

/* Exportar para uso modular */
if (typeof module !== 'undefined') module.exports = MariaVoice;

        now = time.time()
        if key in self._cache:

def filter_anchors(anchors, weather, hours, desires):
    """Filtra âncoras por clima + tempo + desejos. FREE local."""
    budget = hours * 0.85
    kws    = [d.lower() for d in desires]
    scored = []
    for a in anchors:
        s = 0
        if not weather.is_outdoor_ok and a["indoor"]: s += 3
        elif weather.is_outdoor_ok and not a["indoor"]: s += 2
        for kw in kws:
            if kw in a["name"].lower() or kw in a["type"]: s += 5
        scored.append((s, a))

    result, total = [], 0.0
    for _, a in scored:
        if total + a["duration_h"] <= budget:
            result.append(a); total += a["duration_h"]
        if total >= budget: break
    return result

async function callMariaContext(command, lat, lon, lang) {

/* ElevenLabs streaming — narração compacta */
async function narrateViaElevenLabs(text, lang, receiptId) {
  try {
    const secret = sessionStorage.getItem('windi_gateway_secret') || '';
    const params = new URLSearchParams({
      actor: sessionStorage.getItem('windi_travel_did') || 'anonymous',
      caption: text.slice(0, 150),  // narração compacta ~150 chars
      gps: '47.49, 9.46',
      language: lang,
    });
    const resp = await fetch(
      `https://www.windi-domain.com/gateway/voice/stream?${params}`,
      { headers: { 'X-Gateway-Secret': secret } }
    );
    if (!resp.ok) return;
    const blob   = await resp.blob();
    const url    = URL.createObjectURL(blob);
    const audio  = new Audio(url);
    audio.play();
    audio.onended = () => URL.revokeObjectURL(url);
  } catch(e) {
    console.warn('[Maria] ElevenLabs stream:', e);
    // I10 — fallback gracioso: sem áudio, plano já está visível
  }
}
